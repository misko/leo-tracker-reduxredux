def annotate_fold(d, i, full_times, n, seed, plan, *, candidates, phase_step, shifted_origin, on_proposal=None):
    """Evaluate one predeclared grouped split without adopting associations."""
    track = d['scan'].tracks[i]
    if not shifted_origin and len(np.unique(np.floor(track['times'] / 12))) < 3:
        return dict(attempt=dict(status='insufficient_group_support', seed=seed), nomination=None, fold=None)
    train, held, split = planned_split(full_times, track['times'], seed, plan)
    train_target, held_target = (train[n:], held[n:])
    if train_target.sum() < 8 or held_target.sum() < 4:
        return dict(attempt=dict(status='insufficient_group_support', seed=seed, split=split), nomination=None, fold=None)
    keep = d['keep'] & train[:n]
    base = _training_pair(d, d['node'], keep)
    offers, receipt = nominate(d, base, i, train_target, candidates)
    nomination = dict(selected=offers, nomination=receipt, split=split)
    proposals = []
    for offer in offers:
        j = offer['candidate_index']
        norad = int(d['scan'].norads[j])
        bank = {key: d['scan'].sky[key][[j]] for key in ('position_km', 'velocity_km_s')}
        bank['time_s'] = d['scan'].sky['time_s']
        tt = track['times'][train_target]
        frequency, _ = paired_phase_frequency(bank, tt, np.zeros(len(tt), int), np.array([offer['phase_s']]), *d['scan'].search_region.latlon(offer['position']))
        p, x, _ = base['c0']
        alias = int(profile_rate(corrected(p, x, i)[train_target], track['rf_hz'] / 11500000000.0 * frequency, tt)['alias'][0])
        observed = track['values'] + alias * PERIOD
        arms = {}
        for arm, (p, x, meta) in base.items():
            means, cov, lp, phases, elevations, prior = prediction_models(d, p, x, i, j, phase_step=phase_step, uncalibrated_phase_mixture=True)
            if np.isfinite(lp).any():
                score = mixture_prediction(observed, means, cov, lp, train_target, held_target)
                weights = score['training_mixture_weights']
                visible = float(elevations[weights > 1e-06].min())
                phase_mean = float(weights @ phases)
                phase_sd = float(np.sqrt(weights @ (phases - phase_mean) ** 2))
                boundary = float(weights[abs(phases) >= 14.9].sum()) if len(phases) > 1 else float(abs(phases[0]) >= 14.9)
            else:
                score = dict(negative_twice_log_density=1e+30, rms_hz=1000000000000000.0, unobservable=True)
                visible = float(elevations.max())
                phase_mean = phase_sd = boundary = None
            arms[arm] = dict(**meta, scores=dict(conditional_target=score, target=score), prior=prior, minimum_target_elevation_deg=visible, target_phase_training_mean_s=phase_mean, target_phase_training_sigma_s=phase_sd, target_phase_boundary_mass=boundary, target_enters_calibration_equations=False)
        proposal = dict(candidate_index=j, norad=norad, alias=alias, arms=arms)
        proposals.append(proposal)
        if on_proposal is not None:
            on_proposal(seed, proposal)
    fold = dict(seed=seed, split=split, proposals=proposals)
    return dict(attempt=fold, nomination=nomination, fold=fold)
